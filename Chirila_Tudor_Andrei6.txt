//Problem 1
/* program that computes the area and checks if the diagonal  is equal with c with a precision of 2 digits*/
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <math.h>
int main( ){
    int a,b,c;
    const float eps = 0.01;
    printf("Enter three integers: ");
    scanf("%d %d %d", &a, &b, &c);// reading of variables
    float area=(float)a * b; //area formula
    float diag= sqrt((float) a*a + (float) b*b);//diag formula
    printf("Area of rectangle: %.2f\n", area);
    float difference = fabs(diag - c);
    if(difference < eps){
        printf("The diagonal is equal with c.\n");
    } else {
        printf("The diagonal is not equal to c.\n");
    }
    return 0;   
}

//program that reads a number from the KB and finds its greatest prime number and displays all the prime numbers smaller than it
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

int isprime(int n);  //function that checks if a number is prime
   

int main(){
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);
    for(int i=n-1;i>=2;i--){ //loop that takes the number in decreasing order and when it finds the first prime number it displays it and stops
        if(isprime(i)){
            printf("The biggest prime factor of %d is: %d\n", n, i);
            break;
        }
    }
    printf("all prime factors smaller than %d are: ", n);
    for(int i=2;i<n;i++){ //loop that takes all the prime numbers smaller than the number n
        if(isprime(i))
            printf("%d ", i);
    }
    printf("\n");
    return 0;
}

int isprime(int n){
    if(n<2){
        return 0;
    }
    for(int d=2;d<n;d++){
        if(n%d==0){
            return 0;
        }
    }
    return 1;
}

//Problem6
//program that determines the Greatest Common Divider (GCD) of two numbers
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

int gcd(int a, int b);

int main(){
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d", &a);
    scanf("%d", &b);
    printf("The GCD of %d and %d is: %d", a, b, gcd(a, b));
    return 0;
}

int gcd(int a, int b){
    int  r;
    while(b != 0){    //euclid algorithmm
        r = a % b;
        a = b;
        b = r;
    }
    return a;
}

#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

int product(int a, int b);

int main(){
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d", &a);
    scanf("%d", &b);
    printf("The product of %d and %d is: %d", a, b, product(a, b));
    return 0;
}

int product(int a, int b){
    int result = 0;
    for(int i = 0; i < b; i++){
        result += a;
    }
    return result;
}
