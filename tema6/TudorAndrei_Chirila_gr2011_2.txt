//program that checks if a number is a perfect square
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header files used
#include <stdio.h>
#include <math.h>

//Function prototype
int is_integer(float nmb);

int main() {
//local declarations
    int a;
    //reading integer from KB
    printf("Enter an integer: ");
    scanf("%d", &a);

    //condition checking if number is a perfect square
    if (a >= 0 && is_integer(sqrt(a))) {
        printf("%d is a perfect square\n", a);
    } else {
        printf("%d is not a perfect square\n", a);
    }

    return 0;
}

//Function definition

//function that checks if a number is an integer
int is_integer(float nmb) {
    return floor(nmb) == ceil(nmb); //floor and ceil are equal for integers
}
