//program that reads n numbers and shows the one that divide to 3
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header file used
#include <stdio.h>

int main(void) {
//local declarations
    int a, n;
    //reading how many numbers to insert from KB
    printf("Enter how many numbers to insert: ");
    scanf("%d", &n);

    //loop reading n numbers and checking divisibility by 3
    for (int i = 0; i < n; i++) {
        printf("Enter a number: ");
        scanf("%d", &a);
        if (a % 3 == 0) { //condition to divide by 3
            printf("%d divides by 3\n", a);
        }
    }

    return 0;
}
