//Problem 1
//program that computes the area and checks if the diagonal is equal with c with a precision of 2 digits
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header files used
#include <stdio.h>
#include <math.h>

int main(void) {
//local declarations
    int a, b, c;
    const float eps = 0.01f;

    //reading of variables from KB
    printf("Enter three integers: ");
    scanf("%d %d %d", &a, &b, &c); // reading of variables

    //calculating area and diagonal
    float area = (float)a * b; // area formula
    float diag = sqrt((float)a * a + (float)b * b); // diag formula

    //printing the area to terminal
    printf("Area of rectangle: %.2f\n", area);

    //checking if diagonal is equal with c
    float difference = fabs(diag - c);
    if (difference < eps) {
        printf("The diagonal is equal with c.\n");
    } else {
        printf("The diagonal is not equal to c.\n");
    }

    return 0;
}

//Problem 5
//program that reads a number from the KB and finds its greatest prime number and displays all the prime numbers smaller than it
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header file used
#include <stdio.h>

//Function prototype
int isprime(int n); // function that checks if a number is prime

int main(void) {
//local declarations
    int n;
    //reading number from KB
    printf("Enter a number: ");
    scanf("%d", &n);

    //loop that takes the number in decreasing order and when it finds the first prime number it displays it and stops
    for (int i = n - 1; i >= 2; i--) {
        if (isprime(i)) {
            printf("The biggest prime number smaller than %d is: %d\n", n, i);
            break;
        }
    }

    //loop that takes all the prime numbers smaller than the number n
    printf("All prime numbers smaller than %d are: ", n);
    for (int i = 2; i < n; i++) {
        if (isprime(i)) {
            printf("%d ", i);
        }
    }
    printf("\n");

    return 0;
}

//Function definition

//function that checks if a number is prime
int isprime(int n) {
    if (n < 2) { //numbers smaller than 2 are not prime
        return 0;
    }
    //loop checking divisors
    for (int d = 2; d < n; d++) {
        if (n % d == 0) {
            return 0;
        }
    }
    return 1;
}

//Problem 6
//program that determines the Greatest Common Divider (GCD) of two numbers
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header file used
#include <stdio.h>

//Function prototype
int gcd(int a, int b);

int main(void) {
//local declarations
    int a, b;
    //reading two numbers from KB
    printf("Enter two numbers: ");
    scanf("%d", &a);
    scanf("%d", &b);

    //printing the GCD while calling the function
    printf("The GCD of %d and %d is: %d\n", a, b, gcd(a, b));
    return 0;
}

//Function definition

//function that determines the GCD using Euclid algorithm
int gcd(int a, int b) {
//local declaration
    int r;
    //loop for euclidean algorithm
    while (b != 0) { // Euclid algorithm
        r = a % b;
        a = b;
        b = r;
    }
    return a;
}

//Problem 8
//Calculate the product of 2 integer numbers using additions of the first number given by the second one
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header file used
#include <stdio.h>

//Function prototype
int product(int a, int b);

int main(void) {
//local declarations
    int a, b;
    //reading two numbers from KB
    printf("Enter two numbers: ");
    scanf("%d", &a);
    scanf("%d", &b);

    //printing the product while calling the function
    printf("The product of %d and %d is: %d\n", a, b, product(a, b));
    return 0;
}

//Function definition

//function that computes product of two numbers using repeated addition
int product(int a, int b) {
//local declaration
    int result = 0; //result initialization
    //loop adding a, b times
    for (int i = 0; i < b; i++) {
        result += a;
    }
    return result;
}

//Problem 13
//program that determines the Least Common Multiple (LCM) of two numbers
//preprocessor directive
#define _CRT_SECURE_NO_WARNINGS
//header file used
#include <stdio.h>

//Function prototype
int lcm(int a, int b);

int main(void) {
//local declarations
    int a, b;
    //reading two numbers from KB
    printf("Enter two numbers: ");
    scanf("%d", &a);
    scanf("%d", &b);

    //printing the LCM while calling the function
    printf("The LCM of %d and %d is: %d\n", a, b, lcm(a, b));
    return 0;
}

//Function definition

//function that determines the LCM using GCD from Euclid algorithm
int lcm(int a, int b) {
//local declarations
    int c = a, d = b, r;
    //loop for euclidean algorithm
    while (d != 0) { // Euclid algorithm
        r = c % d;
        c = d;
        d = r;
    }
    return (a / c) * b; //LCM formula
}
