//Problem 2
//Program that reads the 3 sides of a triangle, checks if it is a triangle and calculates the area

#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <math.h>

int main( ){
    int a, b, c;
    float s, ar;
    printf("Enter an integer value for all the 3 sides of a triangle: ");
    scanf("%d %d %d", &a, &b, &c);
    if(a > 0 && b > 0 && c > 0 && a + b > c && a + c > b && b + c > a){
        printf("The triangle is valid\n");
        s = (a + b + c) / 2.0;
        ar = sqrt(s * (s - a) * (s - b) * (s - c));
        printf("The values of the sides of the triangle are: %-10.2f %-10.2f %-10.2f\n", (float)a, (float)b, (float)c);
        printf("Its area is: %-10.2f\n", ar);
    }
    else{
        printf("The values %d, %d, %d cannot form a triangle\n", a, b, c);
    }
    return 0;
}
